## 1. 新建 `src/composables/useCollapseTransition.js`

CSS 已负责静态态(`.node-folder.is-collapsed > ul { height: 0 }`、`.node-folder.animating > ul { transition: height }`,均在 `CatalogueTree` 非 scoped 块,doc《目录树》§5 已落地)。本 composable 只负责**两态之间的 height 动画**(移植源 `toggleFolder.js`):测真实高度 → 加 `animating` 启用过渡 → 设终点 → `transitionend` 后把 height 交还 CSS。

```js
import { nextTick, ref, watch } from 'vue'

/**
 * 本 composable 用于文件夹展开 / 折叠的高度过渡(命令式 DOM 副作用)
 * - 机制:折叠从内容高度动到 0,展开从 0 动到内容高度;scrollHeight 始终报告内容高度(不受 height:0 影响)
 * - CSS 负责静态态;本函数只负责过渡。无子节点(file 行)elementRef 为 null,watch 直接返回
 * @param {import('vue').Ref<HTMLElement|null>} elementRef 被动画的 <ul> 元素 ref
 * @param {() => Boolean} isCollapsed 读取当前折叠态的 getter
 * @returns {{ isAnimating: import('vue').Ref<Boolean> }} isAnimating:供组件绑定 .animating 类(启用过渡 + 屏蔽 pointer)
 * */
export function useCollapseTransition(elementRef, isCollapsed) {
    /**
     * @type {import('vue').Ref<Boolean>} 是否正在播放折叠 / 展开动画
     * */
    const isAnimating = ref(false)

    /**
     * 本函数用于播放一次折叠 / 展开的高度过渡
     * @param {Boolean} collapsed 本次切换的目标态:true 折叠 / false 展开
     * */
    async function playTransition(collapsed) {
        const element = elementRef.value
        if (element === null) {
            return
        }

        const startHeight = collapsed ? element.scrollHeight : 0
        const endHeight = collapsed ? 0 : element.scrollHeight

        element.style.height = startHeight + 'px'
        isAnimating.value = true

        // 等 animating 类应用(过渡生效)+ 起点高度落地,再设终点,才会过渡而非跳变
        await nextTick()
        element.offsetHeight // 强制一次 reflow 保险

        element.style.height = endHeight + 'px'

        function onTransitionEnd(event) {
            if (event.propertyName !== 'height') {
                return
            }
            // 交还 CSS:展开后 auto;折叠后清除内联(CSS .is-collapsed > ul 给 0)
            element.style.height = collapsed ? '' : 'auto'
            isAnimating.value = false
            element.removeEventListener('transitionend', onTransitionEnd)
        }
        element.addEventListener('transitionend', onTransitionEnd)
    }

    watch(isCollapsed, playTransition)

    return { isAnimating }
}
```

> **为什么测高度可靠**:`watch` 默认 `flush: 'pre'`,回调在 DOM 应用新 `is-collapsed` 类**之前**跑——折叠时元素仍展开(`scrollHeight` = 内容高度),展开时元素仍折叠(`scrollHeight` 仍报告内容高度)。两向都拿得到正确终点。
>
> **快速连点 / reduced-motion** 的边界(动画被打断、`prefers-reduced-motion` 下跳过动画)留到 **2d** 收口,本步先跑通常规单击。

## 2. 改 `src/components/article/TreeNode.vue`

把静态规则的 `isCollapsed` 换成读 store,`<ul>` 接动画,`.row` 接点击。

```vue
<template>
    <li
        class="tree-node"
        :class="[`node-${node.type}`, {'is-collapsed': isCollapsed, animating: isAnimating}]"
        :style="{'--depth': depth}"
        :data-id="node.id"
    >
        <!-- case1. depth=0:根行(梯形 + 文字,无图标) -->
        <div v-if="depth === 0" class="row primary-row" @click="onRowClick">
            <div class="left-trapezoid"></div>
            <span class="literal">{{node.name}}</span>
        </div>

        <!-- case2. depth>0 文件夹:箭头 + 文字 -->
        <div v-else-if="node.type === CatalogueNodeType.folder" class="row folder-row" @click="onRowClick">
            <i class="iconfont icon-icon-right"></i>
            <span class="literal">{{node.name}}</span>
        </div>

        <!-- case3. depth>0 文件:文件图标 + 文字 -->
        <div v-else class="row file-row" @click="onRowClick">
            <i class="iconfont icon-icon_file"></i>
            <span class="literal">{{node.name}}</span>
        </div>

        <!-- 子节点:递归自身 -->
        <ul v-if="hasChildren" ref="children" class="children">
            <TreeNode
                v-for="child in node.children"
                :key="child.id"
                :node="child"
                :depth="depth + 1">
            </TreeNode>
        </ul>
    </li>
</template>

<script setup>
import {computed, useTemplateRef} from "vue";
import {CatalogueNodeType} from "@/lib/enum.js";
import {useArticleStore} from "@/stores/catalogue.js";
import {useCollapseTransition} from "@/composables/useCollapseTransition.js";

/**
 * @typedef {import('@/data/treeData.js').CatalogueNode} CatalogueNode 目录树节点
 * */

defineOptions({
    name: 'TreeNode',
})

const props = defineProps({
    node: {
        type: /** @type {import('vue').PropType<CatalogueNode>} */ (Object),
        required: true,
    },
    depth: {
        type: Number,
        required: true,
    }
})

/**
 * @type {import('@/stores/catalogue.js').ArticleStore} 文章阅读器状态机的引用
 * */
const articleStore = useArticleStore()

/**
 * @type {import('vue').ComputedRef<Boolean>} 本派生量用于标识节点是否存在子节点
 * */
const hasChildren = computed(() => {
    return Array.isArray(props.node.children) && props.node.children.length > 0
})

/**
 * @type {import('vue').ComputedRef<Boolean>} 本节点(文件夹)是否折叠(读 store)
 * */
const isCollapsed = computed(() => articleStore.isCollapsed(props.node.id))

// 折叠 / 展开的高度动画(命令式 DOM 副作用,封装在 composable)
const childrenRef = useTemplateRef('children')
const {isAnimating} = useCollapseTransition(childrenRef, () => isCollapsed.value)

/**
 * 本函数用于行点击:文件夹切换折叠 + 任意节点选中(忠实源 clickHandle 的耦合)
 * */
function onRowClick() {
    if (props.node.type === CatalogueNodeType.folder) {
        articleStore.toggleCollapsed(props.node.id)
    }
    articleStore.selectNode(props.node.id)
}
</script>
```

要点:
- `isCollapsed` 现在是 `articleStore.isCollapsed(node.id)`:`file` 节点恒 false、`folder` 跟随 store。
- `<ul>` 加 `ref="children"`;`li` 的 `:class` 加 `animating: isAnimating`。
- 源没有"选中行高亮",本步也不加(目录行只有 hover 态);若日后要选中态高亮,再加 `node.id === articleStore.activeNodeId` 的 class。
- ⚠ **根行也会响应点击**:根是 folder,点它会 `toggleCollapsed(根)` 把整棵树折叠(忠实源行为)。是否要让根免折叠,留到 **2d** 定。

## 3. 改 `src/components/article/CatalogueTree.vue`

承接你已做的 `stateClass` computed 重构,把数据源从本地 `phase` 换成 store,删掉本地 `phase` / `RenderPhase` import。

```vue
<script setup>
import {computed, onMounted, useTemplateRef} from "vue";
import SimpleBar from "simplebar";
import TreeNode from "@/components/article/TreeNode.vue";
import {treeData} from "@/data/treeData.js";
import {useArticleStore} from "@/stores/catalogue.js";

defineOptions({
    name: 'CatalogueTree',
})

/**
 * @type {import('@/stores/catalogue.js').ArticleStore} 文章阅读器状态机的引用
 * */
const articleStore = useArticleStore()

/**
 * @type {import('vue').ComputedRef<String>} 目录区当前状态对应的CSS类名
 * */
const stateClass = computed(() => {
    return articleStore.isCatalogueActive ? 'catalogue-active' : 'catalogue-inactive'
})

const rootRef = useTemplateRef('root')

onMounted(() => {
    new SimpleBar(rootRef.value)
})
</script>
```

- 模板不变(仍 `:class="stateClass"`,`<TreeNode :node="treeData" :depth="0">`)。
- 删除:`import {ref} ...` 里的 `ref`(不再用)、`import {RenderPhase} ...`、`const phase = ref(...)`。
- `treeData` 仍要 import(根节点传给 `TreeNode`)。

> `ensureNodeVisible`(openArticle 后滚动)在 **2c** 接(它跟 `pendingRevealId` 一起)。
