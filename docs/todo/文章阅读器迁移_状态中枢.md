## 1. 新建 `src/lib/treeQuery.js`

```js
import { CatalogueNodeType } from '@/lib/enum.js'

/**
 * @typedef {import('@/data/treeData.js').CatalogueNode} CatalogueNode
 * */

/**
 * @typedef {Object} NodeIndexEntry 节点索引项
 * @property {CatalogueNode} node 节点本身
 * @property {CatalogueNode[]} path 根→父 路径(不含自身)
 * */

/**
 * 本函数以 DFS 查找指定 id 的节点及其根→父路径
 * - 自递归:命中处 path 为 [],回溯途中各层把自己前插,天然得到「根→父」顺序(不含命中节点)
 * @param {CatalogueNode} root 树根节点(递归时为当前子树根)
 * @param {Number} targetId 目标节点 id
 * @returns {{node: CatalogueNode|null, path: CatalogueNode[]}} 命中节点 + 路径(不含自身);未命中则 node=null、path=[]
 * */
export function dfsFindNodeWithPath(root, targetId) {
    if (root.id === targetId) {
        return { node: root, path: [] }
    }

    if (root.type === CatalogueNodeType.folder && Array.isArray(root.children)) {
        for (const child of root.children) {
            const hit = dfsFindNodeWithPath(child, targetId)
            if (hit.node !== null) {
                hit.path.unshift(root)
                return hit
            }
        }
    }

    return { node: null, path: [] }
}

/**
 * 本函数查找指定文件夹下 DFS 序的第一个叶子(文件)
 * - 分工:本函数只负责「按 id 定位文件夹 + 校验」,子树内的扫描委托 dfsFirstLeaf
 * @param {CatalogueNode} root 树根节点
 * @param {Number} folderId 文件夹节点 id
 * @returns {CatalogueNode|null} 首个叶子;folderId 非文件夹 / 无子 / 无叶 则 null
 * */
export function dfsFindFirstLeafNode(root, folderId) {
    const { node } = dfsFindNodeWithPath(root, folderId)
    if (node === null || node.type !== CatalogueNodeType.folder || !Array.isArray(node.children)) {
        return null
    }
    return dfsFirstLeaf(node)
}

/**
 * 本函数在给定子树内扫 DFS 序的首个文件节点
 * - 分工:不按 id 定位、不做校验,拿到哪个节点就扫哪棵子树;子树根自身是文件则直接命中返回自身
 * @param {CatalogueNode} node 子树根节点
 * @returns {CatalogueNode|null} DFS 序首个文件节点;子树内无文件则 null
 * */
function dfsFirstLeaf(node) {
    if (node.type === CatalogueNodeType.file) {
        return node
    }
    if (node.type === CatalogueNodeType.folder && Array.isArray(node.children)) {
        for (const child of node.children) {
            const found = dfsFirstLeaf(child)
            if (found !== null) {
                return found
            }
        }
    }
    return null
}

/**
 * 本函数一次 DFS 建立 id→{node, path} 索引,供 O(1) 查节点 / 路径
 * - 显式栈迭代:产物是 Map,与访问顺序无关(pop 使兄弟逆序入索引,不影响结果)
 * @param {CatalogueNode} root 树根节点
 * @returns {Map<Number, NodeIndexEntry>}
 * */
export function buildNodeIndex(root) {
    const nodeIndex = new Map()
    const stack = [{ node: root, path: [] }]

    while (stack.length > 0) {
        const { node, path } = stack.pop()
        nodeIndex.set(node.id, { node, path })
        if (node.type === CatalogueNodeType.folder && Array.isArray(node.children)) {
            const childPath = [...path, node]
            node.children.forEach(child => stack.push({ node: child, path: childPath }))
        }
    }

    return nodeIndex
}

/**
 * 本函数收集初始应折叠的文件夹 id(根展开、depth≥1 的文件夹折叠)
 * @param {CatalogueNode} root 树根节点
 * @returns {Set<Number>}
 * */
export function initCollapsedIdSet(root) {
    const collapsedIdSet = new Set()
    const stack = [{ node: root, depth: 0 }]

    while (stack.length > 0) {
        const { node, depth } = stack.pop()
        if (node.type === CatalogueNodeType.folder && depth >= 1) {
            collapsedIdSet.add(node.id)
        }
        if (Array.isArray(node.children)) {
            node.children.forEach(child => stack.push({ node: child, depth: depth + 1 }))
        }
    }

    return collapsedIdSet
}

/**
 * 本函数取叶子节点的 markdown 正文:有 content 用之,否则长示例兜底(联调滚动 / 样式)
 * @param {CatalogueNode} node 叶子节点
 * @returns {String} markdown 文本
 * */
export function getLeafContent(node) {
    if (typeof node.content === 'string' && node.content.trim() !== '') {
        return node.content
    }
    return buildSampleMarkdown(node)
}

// 接 axios 后由 node.content 取代;此处仅为静态联调兜底(可按源 getLeafMarkdownContent 扩展更长)
function buildSampleMarkdown(node) {
    const sectionList = []
    for (let index = 1; index <= 12; index++) {
        sectionList.push(
            `### 示例章节 ${index}\n\n` +
            `这是第 ${index} 段示例正文,用于验证文章区长内容下的滚动 / 排版,含段落、列表、引用和代码块。\n\n` +
            `- 要点 A:滚动条轨道与滑块渐变\n- 要点 B:内容区与底部按钮的遮挡\n- 要点 C:专注 / 恢复切换后滚动条仍生效\n\n` +
            `> 第 ${index} 段引用,观察块级间距。\n\n` +
            '```js\n' + `const chapter = ${index}\nconsole.log('${node.name}', chapter)\n` + '```\n'
        )
    }

    return `## ${node.name}\n\n这是示例 Markdown 长文档,后续替换为真实正文。\n\n` +
        `### 当前节点\n\n- 名称:${node.name}\n- 创建:${node.createdAt}\n- ID:${node.id}\n\n` +
        sectionList.join('\n\n')
}
```

> `dfsFindNodeWithPath` 为自递归单函数:源那套「祖先栈入/出」改成**回溯时前插**(命中处 `path=[]`,返回途中各层 `unshift` 自己),省掉仅为穿递累加栈而存在的私有辅助函数,对外仍是 `(root, targetId)` 两参。`dfsFindFirstLeafNode` 的辅助 `dfsFirstLeaf` 则保留——那是「按 id 定位文件夹」与「子树内扫首叶」两段不同算法的分工,不是累加器中转。`buildNodeIndex` / `initCollapsedIdSet` 用**显式栈迭代**而非函数内闭包递归:二者产物是 Map / Set,与访问顺序无关,故可用 `pop` 栈实现且签名保持单参。`buildNodeIndex` 一次遍历即得每个节点的 `path`,store 里查路径不再逐次 DFS。命名上,`initCollapsedIdSet` 的 **Set** 与 `buildNodeIndex` 的 **Index** 同位——条目级修饰词 + 集合级头名词,名字预告 `.has()` 而非 `.includes()`;state 侧同概念同名(`collapsedIdSet`,亦即迁移源模块变量的原名)。

## 2. 改 `src/lib/enum.js` + 新建 `src/stores/article.js`

页面任意时刻恒有一栏占宽 70%,store 存的不是「是否专注」布尔,而是「**聚焦谁**」枚举。先在 `src/lib/enum.js` 末尾追加(并在文件尾 export 块加入 `FocusedPane`):

```js
/**
 * @typedef {Object} FocusedPane 本枚举项用于标识文章页当前被聚焦(占宽70%)的面板
 * @property {String} catalogue 聚焦目录树面板(浏览态 / 专注态)
 * @property {String} article 聚焦文章面板(阅读态)
 * */
const FocusedPane = Object.freeze({
    catalogue: 'catalogue',
    article: 'article',
})
```

然后新建 `src/stores/article.js`:

```js
import { defineStore } from 'pinia'
import { computed, ref } from 'vue'
import { RenderPhase, CatalogueNodeType, FocusedPane } from '@/lib/enum.js'
import { treeData } from '@/data/treeData.js'
import { renderMarkdownToHtml } from '@/lib/markdown.js'
import { buildNodeIndex, dfsFindFirstLeafNode, getLeafContent, initCollapsedIdSet } from '@/lib/treeQuery.js'

/**
 * @typedef {import('@/data/treeData.js').CatalogueNode} CatalogueNode
 * */

// 静态桩:整棵树的 id→{node,path} 索引,建一次。接 axios 后数据到达再重建(见 §4 接缝)。
const nodeIndex = buildNodeIndex(treeData)

export const useArticleStore = defineStore('article', () => {
    // PART1. 状态(只存不可推导量)
    /**
     * @type {import('vue').Ref<Number>} 当前选中节点 id(默认根)
     * */
    const activeNodeId = ref(treeData.id)

    /**
     * @type {import('vue').Ref<Set<Number>>} 折叠的文件夹 id 集(Vue 对 Set 有响应式集合处理,add/delete 即触发)
     * */
    const collapsedIdSet = ref(initCollapsedIdSet(treeData))

    /**
     * @type {import('vue').Ref<String>} 当前被聚焦(占宽 70%)的面板:FocusedPane 枚举值,默认聚焦目录(浏览态)
     * */
    const focusedPane = ref(FocusedPane.catalogue)

    /**
     * @type {import('vue').Ref<Number|null>} 待滚入目录视口的节点 id(openArticle 写;CatalogueTree watch 后清空)
     * */
    const pendingRevealId = ref(null)

    // PART2. 派生:选中节点 / 相位 / 路径 / 面包屑
    /**
     * @type {import('vue').ComputedRef<CatalogueNode>} 当前选中节点(兜底根,恒非空)
     * */
    const activeNode = computed(() => {
        const entry = nodeIndex.get(activeNodeId.value)
        if (entry === undefined) {
            return treeData
        }
        return entry.node
    })

    /**
     * @type {import('vue').ComputedRef<String>} 渲染相位(由选中节点是否叶子决定)
     * */
    const renderPhase = computed(() =>
        activeNode.value.type === CatalogueNodeType.file ? RenderPhase.leaf : RenderPhase.nonLeaf
    )

    /**
     * @type {import('vue').ComputedRef<CatalogueNode[]>} 根→当前节点父 的路径
     * */
    const rootToActivePath = computed(() => {
        const entry = nodeIndex.get(activeNodeId.value)
        if (entry === undefined) {
            return []
        }
        return entry.path
    })

    /**
     * @type {import('vue').ComputedRef<String>} 头部面包屑(路径名 + 当前名,'/' 连)
     * */
    const breadcrumb = computed(() => {
        const names = []
        for (const node of rootToActivePath.value) {
            names.push(node.name)
        }
        names.push(activeNode.value.name)
        return names.join('/')
    })

    // PART3. 派生:布局 / 正文(全 class 驱动,无行内样式)
    /**
     * @type {import('vue').ComputedRef<Boolean>} 目录列是否激活(宽):即「聚焦目录」(浏览态 / 专注态)
     * */
    const isCatalogueActive = computed(() => focusedPane.value === FocusedPane.catalogue)

    /**
     * @type {import('vue').ComputedRef<Boolean>} 文章区是否盖蒙版 / 收窄(专注态)
     * Tips: 文章列的 active/inactive class 由 renderPhase 决定(见 ArticlePane);本量只管"专注收窄 + 蒙版 + 整条按钮"
     * */
    const hasMask = computed(() => renderPhase.value === RenderPhase.leaf && focusedPane.value === FocusedPane.catalogue)

    /**
     * @type {import('vue').ComputedRef<String>} 选中叶子的安全 HTML(非叶为空串,省开销)
     * */
    const articleHtml = computed(() =>
        renderPhase.value === RenderPhase.leaf ? renderMarkdownToHtml(getLeafContent(activeNode.value)) : ''
    )

    // getters(带参):判断某文件夹是否折叠(TreeNode 用)
    /**
     * 本函数用于判断指定文件夹当前是否折叠
     * @param {Number} id 文件夹节点 id
     * @returns {Boolean}
     * */
    function isCollapsed(id) {
        return collapsedIdSet.value.has(id)
    }

    // mutations / actions
    /**
     * 本函数用于选中节点,并按节点类型移交焦点:选文件聚焦文章、选文件夹聚焦目录
     * - 专注态下点任意节点,蒙版随焦点移交自然消失(忠实源 clickHandle 清 hasMask)
     * @param {Number} id 目标节点 id
     * */
    function selectNode(id) {
        activeNodeId.value = id
        if (activeNode.value.type === CatalogueNodeType.file) {
            focusedPane.value = FocusedPane.article
        } else {
            focusedPane.value = FocusedPane.catalogue
        }
    }

    /**
     * 本函数用于切换文件夹折叠态(仅改数据,高度动画在组件侧)
     * @param {Number} id 文件夹节点 id
     * */
    function toggleCollapsed(id) {
        if (collapsedIdSet.value.has(id)) {
            collapsedIdSet.value.delete(id)
        } else {
            collapsedIdSet.value.add(id)
        }
    }

    /**
     * 本函数用于强展开一组文件夹(无动画;openArticle 展开路径用)
     * @param {Number[]} ids 文件夹 id 列表
     * */
    function expandFolders(ids) {
        ids.forEach(id => collapsedIdSet.value.delete(id))
    }

    /**
     * 本函数用于"开启文章":当前文件夹下首叶 → 展开路径 → 选中 → 请求滚动
     * */
    function openArticle() {
        const leaf = dfsFindFirstLeafNode(treeData, activeNodeId.value)
        if (leaf === null) {
            return
        }

        // leaf 必在索引中(索引与树同源同时构建),无需判未命中
        const { path } = nodeIndex.get(leaf.id)
        expandFolders(path.map(node => node.id))
        selectNode(leaf.id)
        // 滚动目标 = 叶子的父文件夹(路径末位);叶子必非根,path 至少含根,无需判空
        pendingRevealId.value = path[path.length - 1].id
    }

    /**
     * 本函数用于把焦点交给目录(阅读态 → 专注态;浏览态本就聚焦目录,调用幂等)
     * */
    function focusCatalogue() {
        focusedPane.value = FocusedPane.catalogue
    }

    /**
     * 本函数用于把焦点交还文章(专注态 → 阅读态)
     * - 守卫:聚焦文章仅在 leaf 态合法((nonLeaf, article) 是非法组合,会破两栏布局),非 leaf 调用直接忽略
     * */
    function focusArticle() {
        if (renderPhase.value !== RenderPhase.leaf) {
            return
        }
        focusedPane.value = FocusedPane.article
    }

    /**
     * 本函数用于消费滚动信号(CatalogueTree 滚动后调)
     * */
    function clearPendingReveal() {
        pendingRevealId.value = null
    }

    return {
        // state
        activeNodeId, collapsedIdSet, focusedPane, pendingRevealId,

        // 派生:节点 / 相位 / 路径 / 面包屑
        activeNode, renderPhase, rootToActivePath, breadcrumb,

        // 派生:布局 / 正文
        isCatalogueActive, hasMask, articleHtml,

        // getters(带参)
        isCollapsed,

        // mutations / actions
        selectNode, toggleCollapsed, expandFolders, openArticle,
        focusCatalogue, focusArticle, clearPendingReveal,
    }
})

/**
 * @typedef {ReturnType<typeof useArticleStore>} ArticleStore 文章阅读器状态机 store 实例
 * */
```

> **为什么没有 `isArticleActive`**:文章列的 `article-active`/`article-inactive` class 由 `renderPhase` 决定(leaf→active);专注态文章列**仍是 `.article-active`**(保留梯形标题 + markdown 布局 + 现有蒙版 `.article-active.has-mask::after`),只是用 `.has-mask` 收窄宽度。故无需单独的"文章激活"getter——`renderPhase` + `hasMask` 足矣。(此处细化了《总览》§4,见 2c。)
>
> **为什么 `focusedPane` 是枚举而非布尔(原 `isFocusMode`)**:页面任意时刻恒有一栏 70%,状态的真实语义是「**聚焦谁**」,不是「是否处于某模式」。枚举化后 `isCatalogueActive = focusedPane === catalogue` 直读,原先 `phase===nonLeaf || isFocusMode` 的 `||` 公式消失;阅读⇄专注两个按钮还原为 `focusCatalogue()` / `focusArticle()` 的对称切换。注意枚举只保证**值域**合法,`(nonLeaf, article)` 这种**组合**非法靠转移纪律排除:`selectNode` 双轴一致赋值、`focusCatalogue` 任意相位合法(幂等)、`focusArticle` 带 leaf 守卫。`pendingFocusId` 同步更名 `pendingRevealId`——它的语义是「滚动显形」,把 focus 词汇让给布局轴(也与 `stores/focus.js` 的相机聚焦解耦)。
